   404→                /* ---------- 8) Cart quantity steppers (Step 8 UX glue) ----------
   405→                 * The classic WC cart only ships a bare number input; demo-like
   406→                 * Iranian shops use +/− steppers. Buttons are injected around
   407→                 * the input and drive it through WC's own qty field so the
   408→                 * «بروزرسانی سبد خرید» flow stays intact. */
   409→                [].forEach.call(document.querySelectorAll('.woocommerce-cart table.shop_table .quantity'), function (q) {
   410→                        var input = q.querySelector('input.qty');
   411→                        if (!input || q.querySelector('.tka-step')) { return; }
   412→
   413→                        var mk = function (label, cls) {
   414→                                var b = document.createElement('button');
   415→                                b.type = 'button';
   416→                                b.className = 'tka-step ' + cls;
   417→                                b.textContent = label;
   418→                                b.setAttribute('aria-label', cls === 'minus' ? 'کاهش تعداد' : 'افزایش تعداد');
   419→                                return b;
   420→                        };
   421→
   422→                        var step = function (delta) {
   423→                                var val = parseInt(input.value, 10) || 0;
   424→                                var min = parseFloat(input.getAttribute('min')) || 1;
   425→                                var max = parseFloat(input.getAttribute('max'));
   426→                                var next = val + delta;
   427→                                if (next < min) { next = min; }
   428→                                if (!isNaN(max) && max > 0 && next > max) { next = max; }
   429→                                if (next === val) { return; }
   430→                                input.value = next;
   431→                                /* Same event WC's own handlers listen to. */
   432→                                input.dispatchEvent(new Event('change', { bubbles: true }));
   433→                                /* Classic cart needs the update button; do it silently. */
   434→                                var form = input.closest('form.woocommerce-cart-form');
   435→                                var upd = form ? form.querySelector('button[name="update_cart"]') : null;
   436→                                if (upd) {
   437→                                        upd.disabled = false;
   438→                                        if (window.jQuery) {
   439→                                                window.jQuery(upd).trigger('click');
   440→                                        } else {
   441→                                                upd.click();
   442→                                        }
   443→                                }
   444→                        };
   445→
   446→                        var minus = mk('−', 'minus');
   447→                        var plus = mk('+', 'plus');
   448→                        q.insertBefore(minus, input);
   449→                        q.appendChild(plus);
   450→                });

   720→        function tkaInitStickyHeader() {
   721→                var header = document.querySelector('.header');
   722→                if (!header || !('IntersectionObserver' in window)) return;
   723→                if (header.dataset.tkaSticky) return;
   724→                header.dataset.tkaSticky = '1';
   725→
   726→                var sentinel = document.createElement('div');
   727→                sentinel.setAttribute('aria-hidden', 'true');
   728→                sentinel.style.cssText =
   729→                        'position:relative;width:1px;height:1px;margin:0;padding:0;pointer-events:none';
   730→                header.parentNode.insertBefore(sentinel, header);
   731→
   732→                var io = new IntersectionObserver(function (entries) {
   733→                        for (var i = 0; i < entries.length; i++) {
   734→                                document.body.classList.toggle('tka-stuck', !entries[i].isIntersecting);
   735→                        }
   736→                }, { threshold: 0 });
   737→                io.observe(sentinel);
   738→
   739→                /* TKA 0.17.0 (ممیزی P2-6): توکن پویای ارتفاع هدر.
   740→                   --header-h در demo.css مقدار ثابت کهنهٔ 104px دارد؛
   741→                   اینجا ارتفاع «واقعی» هدر (و تغییرش در حالت فشردهٔ
   742→                   .tka-stuck یا ریسپانسیو) با ResizeObserver روی
   743→                   <html> نوشته می‌شود تا هر قانون CSS که به
   744→                   var(--header-h) تکیه می‌کند همیشه درست باشد. */
   745→                if ('ResizeObserver' in window) {
   746→                        var setH = function () {
   747→                                var h = header.offsetHeight;
   748→                                if (h > 0) document.documentElement.style.setProperty('--header-h', h + 'px');
   749→                        };
   750→                        setH();
   751→                        new ResizeObserver(setH).observe(header);
   752→                }
   753→        }
   754→        if (document.readyState === 'loading') {
   755→                document.addEventListener('DOMContentLoaded', tkaInitStickyHeader);
   756→        } else {
   757→                tkaInitStickyHeader();
   758→        }
